// ==== Ultrasonic EXTI / timing capture ====
#include "main.h"
#include "tim6_pulse_capt.h"
#include "sensors.h"

// Adjust to your setup
extern TIM_HandleTypeDef htim6;   // microsecond timer

// Public (extern in .h) for the tester to read
volatile uint8_t  usCaptureComplete = 0;
volatile uint16_t usPulse_us        = 0;
volatile uint32_t lastEchoMs        = 0;

//Local variables
static volatile uint8_t  isRisingCaptured = 0;
static volatile uint16_t usStartTick6     = 0;
static volatile uint32_t riseTickMs       = 0;

//===== START: for debugging for range for ultrasonic =====

// ---- config  ----
#define US_MIN_PULSE_US     80u      // ~1.4 cm
#define US_MAX_PULSE_US   5000u      // ~86 cm (tight while benching 25 cm)
#define US_ARM_WINDOW_MS    40u      // must see both edges within 40 ms of TRIG
#define US_PAIR_WINDOW_MS   10u      // stop edge must arrive within 10 ms of start

// ---- telemetry for debugging ----
volatile uint32_t us_exti_rise_cnt = 0;
volatile uint32_t us_exti_fall_cnt = 0;
volatile uint32_t us_exti_badwin   = 0;  // fell too late
volatile uint32_t us_exti_badrng   = 0;  // delta out of range
volatile uint16_t us_last_delta    = 0;  // last measured (even if rejected)
volatile uint8_t  us_last_reason   = 0;  // 0=ok, 1=late, 2=range

// ---- timing state ----
static volatile uint8_t  edge_pending   = 0;  // 0=waiting first edge, 1=have start, waiting end
static volatile uint16_t t_start        = 0;
static volatile uint32_t t_start_ms     = 0;
static volatile uint32_t trig_ms      = 0;
static volatile uint8_t  us_armed     = 0;   // <— only accept edges while armed

//===== END: for debugging for range for ultrasonic =====

// Call this once after clocks are up
void us_pulse_capture_init(void) {
    // Ensure TIM6 runs free at 1MHz
    HAL_TIM_Base_Start(&htim6);

    // Configure EXTI on PC12 for RISING+FALLING in CubeMX, or here:
    // - GPIOC PC12 input, no pull (depends on sensor)
    // - EXTI line 12, rising & falling edge, NVIC enabled
    // If already configured in CubeMX, you don't need code here.
}

// EXTI callback from HAL (called for any EXTI line)
// Use EXTI on PC12 to time the ultrasonic echo with TIM6 (1us ticks)
void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)
{
    if (GPIO_Pin != US_DATA_Pin) return;

    uint32_t now_ms = HAL_GetTick();
    if (!us_armed) return;                               // ignore strays
    if ((now_ms - trig_ms) > US_ARM_WINDOW_MS) {         // too late for this ping
        us_armed = 0;
        edge_pending = 0;
        return;
    }

    if (!edge_pending) {
        // First edge after TRIG: start timing
        edge_pending = 1;
        t_start      = us_timer_cnt();
        t_start_ms     = now_ms;
        return;
    }

    // Second edge: only accept if close enough to first edge
    if ((now_ms - t_start_ms) > US_PAIR_WINDOW_MS) {       // stale pair → reset for safety
        edge_pending = 0;
        return;
    }

    uint16_t t_end  = us_timer_cnt();
    uint16_t delta  = (t_end >= t_start) ? (t_end - t_start)
                                         : (uint16_t)(65536u - t_start + t_end);

    if (delta >= US_MIN_PULSE_US && delta <= US_MAX_PULSE_US) {
        usPulse_us        = delta;                       // 1 tick = 1 us (TIM6 @ 1 MHz)
        lastEchoMs        = now_ms;
        usCaptureComplete = 1;
        // Feed filter/conversion used by the rest of the system
        sensors_read_usDist((float)delta * 1e-6f);
    }
    // done with this ping regardless
    us_armed = 0;
    edge_pending = 0;
}

// Helper: read TIM6 CNT (1 tick = 1 us)
uint16_t us_timer_cnt(void) {
    return (uint16_t)__HAL_TIM_GET_COUNTER(&htim6);
}

//===== getter functions =====
uint8_t us_is_armed(void)          { return us_armed; }
uint8_t us_edge_pending(void)      { return edge_pending; }
uint32_t us_last_trig_ms(void)     { return trig_ms; }
uint32_t us_last_start_ms(void)    { return t_start_ms; }

//===== setter functions =====
void us_arm_for_trig(void) {
    us_armed     = 1;
    edge_pending = 0;
    trig_ms      = HAL_GetTick();
}
