import os, time, cv2
from typing import Tuple, Any, Optional

# ===== ENV =====
YOLO_WEIGHTS   = os.getenv("YOLO_WEIGHTS", "weights/yolo_updated.pt")
YOLO_CONF      = float(os.getenv("YOLO_CONF", "0.30"))
YOLO_IOU       = float(os.getenv("YOLO_IOU",  "0.45"))
YOLO_IMGSZ     = int(os.getenv("YOLO_IMGSZ",  "640"))
YOLO_DEVICE    = os.getenv("YOLO_DEVICE", "cpu")
YOLO_VERBOSE   = int(os.getenv("YOLO_VERBOSE", "0"))
SAVE_DEBUG     = int(os.getenv("SAVE_DEBUG",   "0"))

# Lưu tất cả detection (kể cả NA) vào cùng một thư mục
DETECT_LOG_DIR = os.getenv("DETECT_LOG_DIR", "det_logs")

# Tiền xử lý
USE_CLAHE      = int(os.getenv("USE_CLAHE", "1"))
MAX_SIDE       = int(os.getenv("MAX_SIDE", "0"))

# GIỮ để không phá các ENV khác đang dùng
ROI_MARGIN     = float(os.getenv("ROI_MARGIN", "0.18"))  # (không dùng trực tiếp)

print("="*60)
print(f"YOLO_DEVICE: {YOLO_DEVICE}")


# ===== Globals =====
_YOLO_MODEL = None
_VALID_SET = {"left", "right", "bullseye"}  # đúng 3 lớp

def _log(*a):
    if YOLO_VERBOSE:
        print(*a, flush=True)

def _ensure_dirs():
    if SAVE_DEBUG:
        os.makedirs(DETECT_LOG_DIR, exist_ok=True)

def _clahe_if_needed(img_bgr):
    if not USE_CLAHE:
        return img_bgr
    yuv = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2YUV)
    y, u, v = cv2.split(yuv)
    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8,8))
    y2 = clahe.apply(y)
    yuv2 = cv2.merge([y2, u, v])
    return cv2.cvtColor(yuv2, cv2.COLOR_YUV2BGR)

def _short_side_resize(img, max_side:int) -> Any:
    if max_side <= 0:
        return img
    h, w = img.shape[:2]
    s = max(h, w)
    if s <= max_side:
        return img
    r = max_side / float(s)
    nh, nw = int(round(h*r)), int(round(w*r))
    return cv2.resize(img, (nw, nh), interpolation=cv2.INTER_AREA)

def _warmup_model(model):
    """Run warmup predictions to initialize the model and speed up first real detection."""
    import numpy as np
    print("[YOLO] Warming up model with dummy predictions...")
    
    # Create a dummy image similar to what we'll process
    dummy_img = np.zeros((640, 640, 3), dtype=np.uint8)
    # Add some content so it's not completely black
    cv2.rectangle(dummy_img, (200, 200), (400, 400), (255, 255, 255), -1)
    cv2.circle(dummy_img, (320, 320), 50, (128, 128, 128), -1)
    
    # Run 3 warmup predictions to fully initialize the model
    for i in range(3):
        _ = model.predict(
            dummy_img, 
            imgsz=YOLO_IMGSZ, 
            conf=YOLO_CONF, 
            iou=YOLO_IOU,
            device=YOLO_DEVICE, 
            verbose=False
        )
    
    print("[YOLO] Warmup complete! First real detection should be fast now.")

def _load_model():
    """Lazy-load YOLOv11 model (Ultralytics)."""
    global _YOLO_MODEL
    if _YOLO_MODEL is not None:
        return _YOLO_MODEL
    if not os.path.isfile(YOLO_WEIGHTS):
        raise FileNotFoundError(f"[YOLO] Weights not found: {YOLO_WEIGHTS}")
    
    # Configure PyTorch to allow loading YOLO weights safely
    import torch
    try:
        # For PyTorch 2.6+, add safe globals for Ultralytics classes
        torch.serialization.add_safe_globals([
            'ultralytics.nn.tasks.SegmentationModel',
            'ultralytics.nn.tasks.DetectionModel',
            'ultralytics.nn.tasks.ClassificationModel',
            'ultralytics.nn.tasks.PoseModel',
            'ultralytics.nn.tasks.OBBModel'
        ])
    except:
        pass  # Older PyTorch versions don't have this
    
    from ultralytics import YOLO
    _YOLO_MODEL = YOLO(YOLO_WEIGHTS)
    _log(f"[YOLO] Loaded weights: {YOLO_WEIGHTS}")
    _log(f"[YOLO] Loaded devices: {YOLO_DEVICE}")
    
    # Warmup the model to speed up first detection
    _warmup_model(_YOLO_MODEL)

    return _YOLO_MODEL

def _draw_box(frame_bgr, box_xyxy, label:str, conf:float) -> Any:
    """Draw bounding box and label on frame with enhanced visibility."""
    out = frame_bgr.copy()
    if box_xyxy is not None:
        x1, y1, x2, y2 = map(int, box_xyxy)
        
        # Draw bounding box with thick green line
        cv2.rectangle(out, (x1, y1), (x2, y2), (0, 255, 0), 3)
        
        # Create label text with confidence
        text = f"{label} {conf:.2f}"
        
        # Use larger font for better readability
        font_scale = 0.7
        thickness = 2
        font = cv2.FONT_HERSHEY_SIMPLEX
        
        # Get text size for background
        (text_width, text_height), baseline = cv2.getTextSize(text, font, font_scale, thickness)
        
        # Calculate text position (above the box, or below if not enough space)
        text_x = x1
        text_y = y1 - 10
        
        # If text would go above image, put it below the box
        if text_y - text_height < 0:
            text_y = y2 + text_height + 10
        
        # Draw semi-transparent background for text
        overlay = out.copy()
        cv2.rectangle(overlay, 
                     (text_x - 5, text_y - text_height - 5), 
                     (text_x + text_width + 5, text_y + baseline + 5), 
                     (0, 255, 0), -1)
        
        # Blend the overlay with the original image for transparency
        alpha = 0.8
        cv2.addWeighted(overlay, alpha, out, 1 - alpha, 0, out)
        
        # Draw text with black color for better contrast
        cv2.putText(out, text, (text_x, text_y),
                    font, font_scale, (0, 0, 0), thickness, cv2.LINE_AA)
        
        # Add a small circle at the center of the detection for precise location
        center_x = (x1 + x2) // 2
        center_y = (y1 + y2) // 2
        cv2.circle(out, (center_x, center_y), 4, (0, 255, 0), -1)
        cv2.circle(out, (center_x, center_y), 4, (0, 0, 0), 1)
    else:
        # NA: no box → write text in corner with yellow background
        text = f"{label} 0.00"
        font_scale = 0.9
        thickness = 2
        font = cv2.FONT_HERSHEY_SIMPLEX
        (tw, th), bl = cv2.getTextSize(text, font, font_scale, thickness)
        pad = 10
        overlay = out.copy()
        cv2.rectangle(overlay, (pad, pad), (pad+tw+10, pad+th+10), (0, 255, 255), -1)
        cv2.addWeighted(overlay, 0.7, out, 0.3, 0, out)
        cv2.putText(out, text, (pad+5, pad+th+5), font, font_scale, (0, 0, 0), thickness, cv2.LINE_AA)
    return out

def _save_det_image(frame_bgr, label:str, conf:float, box_xyxy):
    """Lưu ảnh detection (kể cả NA) vào cùng thư mục DETECT_LOG_DIR."""
    if not SAVE_DEBUG:
        return
    os.makedirs(DETECT_LOG_DIR, exist_ok=True)
    ts = time.strftime("%Y%m%d_%H%M%S")
    fname = f"{ts}_{label}_{conf:.2f}.jpg"
    out = _draw_box(frame_bgr, box_xyxy, label, conf)
    cv2.imwrite(os.path.join(DETECT_LOG_DIR, fname), out)

def _pick_best_from_ultra(result) -> Optional[Tuple[str, float, Tuple[int,int,int,int]]]:
    """Chọn detection tốt nhất trong 3 lớp hợp lệ."""
    try:
        names = result.names  # dict
        if result.boxes is None or len(result.boxes) == 0:
            return None
        best = None
        for b in result.boxes:
            cls_idx = int(b.cls.item())
            label = str(names.get(cls_idx, str(cls_idx)))
            if label not in _VALID_SET:
                continue
            conf = float(b.conf.item())
            x1, y1, x2, y2 = map(int, b.xyxy[0].tolist())
            if (best is None) or (conf > best[1]):
                best = (label, conf, (x1,y1,x2,y2))
        return best
    except Exception as e:
        _log("[pick_best error]", repr(e))
        return None

def detect_symbol(frame_bgr, save_debug: bool=False) -> Tuple[str, float]:
    """
    Trả về (label, confidence), label ∈ {left, right, bullseye, NA}
    Luôn có thể lưu ảnh detection (kể cả NA) vào cùng thư mục nếu SAVE_DEBUG=1 hoặc save_debug=True.
    """
    _ensure_dirs()
    t0 = time.time()

    model = _load_model()
    img = _clahe_if_needed(frame_bgr)
    img = _short_side_resize(img, MAX_SIDE)

    results = model.predict(
        img, imgsz=YOLO_IMGSZ, conf=YOLO_CONF, iou=YOLO_IOU,
        device=YOLO_DEVICE, verbose=False
    )
    result = results[0]
    best = _pick_best_from_ultra(result)

    if not best:
        label, conf, box = "NA", 0.0, None
        if save_debug or SAVE_DEBUG:
            _save_det_image(frame_bgr, label, conf, box)
        dt = 1000*(time.time()-t0)
        print(f"[DETECT] {label} conf={conf:.3f} in {dt:.1f} ms")
        return (label, conf)

    label, conf, box = best
    if save_debug or SAVE_DEBUG:
        _save_det_image(frame_bgr, label, conf, box)

    dt = 1000*(time.time()-t0)
    print(f"[DETECT] {label} conf={conf:.3f} in {dt:.1f} ms")
    return (label, conf)

def detect_symbol_with_annotation(frame_bgr, save_debug: bool=False) -> Tuple[str, float, Any]:
    """
    Trả về (label, confidence, annotated_frame)
    Ảnh annotated cũng lưu chung thư mục (kể cả NA) nếu bật SAVE_DEBUG hoặc save_debug=True.
    """
    _ensure_dirs()
    t0 = time.time()

    model = _load_model()
    img = _clahe_if_needed(frame_bgr)
    img = _short_side_resize(img, MAX_SIDE)

    results = model.predict(
        img, imgsz=YOLO_IMGSZ, conf=YOLO_CONF, iou=YOLO_IOU,
        device=YOLO_DEVICE, verbose=False
    )
    result = results[0]
    best = _pick_best_from_ultra(result)

    if not best:
        label, conf, box = "NA", 0.0, None
        annotated = _draw_box(frame_bgr.copy(), None, label, conf)
        if save_debug or SAVE_DEBUG:
            _save_det_image(frame_bgr, label, conf, None)
        dt = 1000*(time.time()-t0)
        print(f"[DETECT] {label} conf={conf:.3f} in {dt:.1f} ms")
        return (label, conf, annotated)

    label, conf, box = best
    annotated = _draw_box(frame_bgr.copy(), box, label, conf)
    if save_debug or SAVE_DEBUG:
        _save_det_image(frame_bgr, label, conf, box)

    dt = 1000*(time.time()-t0)
    print(f"[DETECT] {label} conf={conf:.3f} in {dt:.1f} ms")
    return (label, conf, annotated)
