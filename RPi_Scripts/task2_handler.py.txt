3. Snap picture and get detection (LEFT/RIGHT/BULLSEYE)
4. Based on detection, send appropriate command set
5. Wait for "fS" after each command in the set
6. Repeat for second obstacle

Features:
- Detects LEFT/RIGHT arrows and BULLSEYE symbols
- Sends movement commands to STM32 based on detections
- Provides live video feed with detection visualization

        try:
            while self.live_feed_active and self.task2_active:

                        break
                
                time.sleep(0.1)
                
        except Exception as e:
            logger.error("Live feed error: {}".format(e))
        finally:
            self.live_feed_active = False
            if cap:
                cap.release()
            cv2.destroyAllWindows()
            
    def _add_overlay(self, frame):
        """Add detection overlay to frame"""
        if not CV2_AVAILABLE:
            return frame

        cv2.putText(display_frame, "TASK 2 - FASTEST CAR", (10, 30), 
                   cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)
        
        # Current detection
        if self.current_detection:
            text = "DETECTED: {}".format(self.current_detection)
            color = (0, 0, 255) if self.current_detection == "BULLSEYE" else (255, 0, 0)
            cv2.putText(display_frame, text, (10, 70), 
                       cv2.FONT_HERSHEY_SIMPLEX, 0.8, color, 2)
        
        # Progress
        progress = "Obstacles: {}/{}".format(self.obstacles_completed, self.total_obstacles)
        cv2.putText(display_frame, progress, (10, 110), 
                   cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 0), 2)
        
        # Detection box
        if self.current_detection and self.current_detection != "NO_DETECTION":
            h, w = display_frame.shape[:2]
            center_x, center_y = w//2, h//2
            cv2.rectangle(display_frame, (center_x-60, center_y-40), 
                         (center_x+60, center_y+40), (0, 255, 255), 3)
            cv2.putText(display_frame, self.current_detection, (center_x-50, center_y-50), 
                       cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 255), 2)
        
        return display_frame
        
    def _run_mission(self):
        """Main mission logic with new obstacle flow"""
        try:
            logger.info("Starting Task 2 navigation sequence...")
            # --- 1st Obstacle ---
            logger.info("Phase 1: Detecting 1st obstacle...")
            self._send_status_to_android("Moving to Obstacle 1...")
            self._send_stm32_command("W50|0|40\n")
            self._send_stm32_command("M\n")
            if self._wait_for_stm32_response():
                self._send_status_to_android("Detecting Obstacle 1...")
                detection1 = self._detect_symbol()
                if detection1 == "LEFT":
                    logger.info("Running 1st LEFT sequence...")
                    self._send_status_to_android("Obstacle 1: LEFT detected")
                    self._run_command_sequence(self._get_1st_left_commands())
                elif detection1 == "RIGHT":
                    logger.info("Running 1st RIGHT sequence...")
                    self._send_status_to_android("Obstacle 1: RIGHT detected")
                    self._run_command_sequence(self._get_1st_right_commands())
                elif detection1 == "BULLSEYE":
                    print("DETECTED BULLSEYE")  # Just print and continue
                    logger.info("Bullseye detected - continuing mission")
                    self._send_status_to_android("Obstacle 1: BULLSEYE detected")
                else:
                    logger.error("No valid detection for 1st obstacle!")
                    self._send_status_to_android("Obstacle 1: Detection failed")
                    self._emergency_stop()
                    return
                self.obstacles_completed += 1
                self._send_status_to_android("Obstacle 1: Completed")

            # --- 2nd Obstacle ---
            logger.info("Phase 2: Detecting 2nd obstacle...")
            self._send_status_to_android("Moving to Obstacle 2...")
            self._send_stm32_command("W50|0|20\n")
            self._send_stm32_command("t50|0|25\n")
            self._send_stm32_command("M\n")
            if self._wait_for_stm32_response():
                self._send_status_to_android("Detecting Obstacle 2...")
                detection2 = self._detect_symbol()
                if detection2 == "LEFT":
                    logger.info("Running 2nd LEFT sequence...")
                    self._send_status_to_android("Obstacle 2: LEFT detected")
                    self._run_command_sequence(self._get_2nd_left_commands())
                elif detection2 == "RIGHT":
                    logger.info("Running 2nd RIGHT sequence...")
                    self._send_status_to_android("Obstacle 2: RIGHT detected")

                else:
                    logger.error("No valid detection for 2nd obstacle!")
                    self._send_status_to_android("Obstacle 2: Detection failed")
                    self._emergency_stop()
                    return
                self.obstacles_completed += 1
                self._send_status_to_android("Obstacle 2: Completed")

            # Mission complete
            self._mission_complete()
            self._send_status_to_android("Task 2 Completed!")
        except Exception as e:
            logger.error("Mission error: {}".format(e))
            self._send_status_to_android("Mission Error: {}".format(str(e)))
            self._emergency_stop()
            
    def _handle_obstacle(self, obstacle_num: int) -> bool:
        """Handle obstacle detection and navigation"""
        logger.info("Handling obstacle {}...".format(obstacle_num))
        
        # Snap picture and perform detection
        detection = self._detect_symbol()
        
        if detection == "LEFT":
            print("DETECTED LEFT")
            logger.info("LEFT arrow detected on obstacle {}".format(obstacle_num))
            self.current_detection = "LEFT"
            return self._execute_left_commands()
            
        elif detection == "RIGHT":
            print("DETECTED RIGHT")
            logger.info("RIGHT arrow detected on obstacle {}".format(obstacle_num))
            self.current_detection = "RIGHT"
            return self._execute_right_commands()
            
        elif detection == "BULLSEYE":
            print("DETECTED BULLSEYE")
            logger.info("BULLSEYE detected - continuing mission")
            self.current_detection = "BULLSEYE"
            return True  # Continue mission
            
        else:
            logger.error("No clear detection from vision system - cannot proceed safely")
            self.current_detection = "NO_DETECTION"
            logger.error("Task 2 requires valid detection to navigate safely")
            self._emergency_stop()
            return False
            
    def _detect_symbol(self) -> str:
        """Detect arrow/bullseye symbol using actual vision system"""
        try:
            # Use actual vision system (same as MDP handler)
            if self.vision_enabled and self.vision_client:
                logger.info("Using actual vision detection for Task 2...")
                
                # Capture and detect single image (same method as MDP handler)
                detection_result = self.vision_client.capture_and_detect_single()
                
                if detection_result and detection_result not in ("", "NO_TEXT", "DECODE_ERROR", "NA"):
                    logger.info("Vision server result: {}".format(detection_result))
                    
                    # Map vision server response to Task 2 symbols
                    detection_upper = detection_result.upper()
                    
                    # Check for arrow symbols
                    if "LEFT" in detection_upper or "L" == detection_upper or "←" in detection_upper:
                        logger.info("Mapped to LEFT arrow")
                        return "LEFT"
                    elif "RIGHT" in detection_upper or "R" == detection_upper or "→" in detection_upper:
                        logger.info("Mapped to RIGHT arrow") 
                        return "RIGHT"
                    elif "BULL" in detection_upper or "TARGET" in detection_upper or "⊙" in detection_upper:
                        logger.info("Mapped to BULLSEYE")
                        return "BULLSEYE"
                    else:
                        logger.info("No recognized symbol in: {}".format(detection_result))
                        return "NO_DETECTION"
                else:
                    logger.info("No valid detection from vision server")
                    return "NO_DETECTION"
            
            else:
                logger.error("Vision system not available! Cannot perform detection.")
                return "NO_DETECTION"
            
        except Exception as e:
            logger.error("Detection error: {}".format(e))
            return "NO_DETECTION"

    def _bluetooth_listen_thread(self):
        """Bluetooth listening thread for Android commands"""
        logger.info("Starting Bluetooth listener for Android Arena commands...")
        
        try:
            # Check if Bluetooth device exists
            if os.path.exists(self.bluetooth_device):
                logger.info("Bluetooth device found: {}".format(self.bluetooth_device))
                logger.info("Ready to receive commands from Android...")
                
                while self.bluetooth_enabled:
                    try:
                        # Open file fresh each time to read new command
                        logger.debug("Opening Bluetooth device to check for command...")
                        with open(self.bluetooth_device, 'r') as bt_conn:
                            # Check if data is available to read (non-blocking)
                            if select.select([bt_conn], [], [], 1.0)[0]:
                                logger.info("Data available on Bluetooth, reading...")
                                # Try to read with a shorter timeout
                                # Read line or partial line
                                try: ## "STM:SP"
                                    command = bt_conn.read(1)  # Read up to 100 chars instead of readline
                                    #command = command.strip()
                                    logger.info("Raw command received: '{}'".format(repr(command)))
                                    if command: 
                                        logger.info("Received Android command: {}".format(command))
                                        self._handle_bluetooth_command(command)
                                        break
                                    else:
                                        logger.debug("Empty command received")
                                except Exception as read_err:
                                    logger.error("Error reading command: {}".format(read_err))
                        
                    except Exception as e:
                        logger.error("Error reading Bluetooth data: {}".format(e))
                        time.sleep(1)
            else:
                logger.warning("Bluetooth device {} not found - using fallback method".format(self.bluetooth_device))
                
               
        except Exception as e:
            logger.error("Bluetooth listener error: {}".format(e))
            logger.info("Bluetooth communication disabled")
            self.bluetooth_enabled = False
        
        logger.info("Bluetooth listener thread ended")
    
    def _handle_bluetooth_command(self, command: str):
        """Handle commands received from Android via Bluetooth"""
        command = command.upper().strip()
        logger.info("Processing Bluetooth command from Android: '{}'".format(command))
        logger.debug("Command length: {}, repr: {}".format(len(command), repr(command)))
        
        # Handle various Android Arena command formats
        if (command == "S" or command == "START" or "TASK2:START" in command or 
            command == "STM:SP" or "SP" in command or "START_PATH" in command):
            logger.info("Received START command from Android Arena (SP Button pressed)")
            print("[ANDROID] SP Button pressed - Starting Task 2...")
            self._send_status_to_android("Task 2 Starting...")
            success = self.start_task2_mission()
            if success:
                self._send_status_to_android("Task 2 Started Successfully")
                print("[ANDROID] Task 2 mission started!")
            else:
                self._send_status_to_android("Task 2 Start Failed - Check connections")
                print("[ANDROID] Failed to start Task 2 mission!")
                
        elif (command == "E" or command == "STOP" or command == "EMERGENCY" or 
              "EMERGENCY" in command or "STOP" in command):
            logger.info("Received EMERGENCY STOP command from Android Arena")
            print("[ANDROID] Emergency stop requested...")
            self._send_status_to_android("Emergency Stop Activated")
            self.stop_mission()
            self._send_status_to_android("Task 2 Stopped")
            print("[ANDROID] Task 2 stopped!")
            
        elif command == "STATUS" or "STATUS" in command:
            status = "Running" if self.task2_active else "Ready"
            progress = "Obstacles: {}/{}".format(self.obstacles_completed, self.total_obstacles)
            detection = " | Detection: {}".format(self.current_detection) if self.current_detection else ""
            full_status = "{} - {}{}".format(status, progress, detection)
            self._send_status_to_android(full_status)
            print("[ANDROID] Status sent: {}".format(full_status))
            
        elif "CONNECT" in command or "INIT" in command:
            logger.info("Android Arena connection established")
            self._send_status_to_android("Task 2 Handler Ready")
            print("[ANDROID] Connected to Arena - Task 2 ready!")
            
        else:
            logger.warning("Unknown Android command: {}".format(command))
            print("[ANDROID] Unknown command received: {}".format(command))
            # Don't send unknown command response to avoid spam
    
    def _send_bluetooth_message(self, message: str):
        """Send message to Android via existing Bluetooth connection"""
        logger.info("Sending to Android: {}".format(message))
        
        try:
            # Try to send via Bluetooth device
            if self.bluetooth_enabled and os.path.exists(self.bluetooth_device):
                with open(self.bluetooth_device, 'w') as bt_conn:
                    bt_conn.write(message + '\n')
                    bt_conn.flush()
                logger.info("Message sent to Android successfully")
            else:
                # Fallback: just log the message
                logger.info("Bluetooth not available - message logged: {}".format(message))
                
        except Exception as e:
            logger.error("Error sending Bluetooth message: {}".format(e))
            logger.info("Message for Android (not sent): {}".format(message))
    
    def _send_status_to_android(self, status_msg: str):
        """Send status update to Android Arena in proper format"""
        # Android Arena expects: STATUS,<message>
        formatted_msg = "STATUS,{}".format(status_msg)
        self._send_bluetooth_message(formatted_msg)
            
    def _send_stm32_command(self, command: str):
        """Send raw command to STM32"""
        if not self.stm32_serial:
            logger.error("STM32 not connected!")
            return False
            
        try:
            logger.info("Sending to STM32: {}".format(command.strip()))
            self.stm32_serial.write(command.encode())
            self.stm32_serial.flush()
            return True
            
        except Exception as e:
            logger.error("Command send error: {}".format(e))
            return False
    
    def _wait_for_stm32_response(self, expected_response="fM", timeout=30):
        """Wait for specific response from STM32"""
        if not self.stm32_serial:
            logger.error("STM32 not connected!")
            return False
            
        logger.info("Waiting for STM32 response: {}".format(expected_response))
        start_time = time.time()
        
        try:
            while time.time() - start_time < timeout:
                if self.stm32_serial.in_waiting > 0:
                    response = self.stm32_serial.readline().decode().strip()
                    logger.info("STM32 response: {}".format(response))
                    
                    if expected_response in response:
                        logger.info("Got expected response from STM32")
                        return True
                
                time.sleep(0.1)
            
            logger.error("Timeout waiting for STM32 response")
            return False
            
        except Exception as e:
            logger.error("Error waiting for STM32 response: {}".format(e))
            return False
    
    def _run_command_sequence(self, commands: list):
        """Run a sequence of STM32 commands, wait for fM after sequence"""
        for cmd in commands:
            if not self.task2_active:
                logger.warning("Task stopped during command sequence.")
                return False
            self._send_stm32_command(cmd)
        # Wait for fM after sequence
        if not self._wait_for_stm32_response(expected_response="fM"):
            logger.error("Did not receive fM after command sequence!")
            return False
        return True

    def _emergency_stop(self):
        """Emergency stop"""
        logger.warning("EMERGENCY STOP ACTIVATED!")
        self.task2_active = False
        self._send_status_to_android("Emergency Stop")
        
        if self.stm32_serial:
            try:
                self._send_stm32_command("S\n")  # Use new command method
                logger.info("Stop command sent to STM32")
            except Exception as e:
                logger.error("Error sending stop: {}".format(e))
                
    def _mission_complete(self):
        """Mission completion"""
        logger.info("=" * 50)
        logger.info("TASK 2 MISSION COMPLETED!")
        logger.info("Obstacles completed: {}/{}".format(self.obstacles_completed, self.total_obstacles))
        logger.info("=" * 50)
        
        self.task2_active = False
        
        # Final stop
        self._send_stm32_command("S\n")
        
        # Signal that mission is complete and program should exit
        self.mission_complete_flag = True
        logger.info("Mission complete - shutting down...")
        
    def stop_mission(self):
        """Stop current mission"""
        logger.info("Stopping Task 2 mission...")
        self._emergency_stop()
        
    def cleanup(self):
        """Clean up resources"""
        logger.info("Cleaning up Task 2 handler...")
        
        self.task2_active = False
        
        if self.stm32_serial:
            try:
                self._send_stm32_command("S\n")
                time.sleep(0.5)
                self.stm32_serial.close()
            except:
                pass
                
        if self.vision_client:
            try:
                self.vision_client.cleanup()
                logger.info("Task 2 vision client cleaned up")
            except:
                pass
        
        if self.bluetooth_conn:
            try:
                self._send_bluetooth_message("DISCONNECTING")
                self.bluetooth_conn.close()
            except:
                pass
                
        self.bluetooth_enabled = False
            
        logger.info("Task 2 cleanup completed")

        handler = Task2Handler()
        
        try:
            handler.initialize()
            
            # Start Bluetooth listening for Android commands
            if handler.bluetooth_enabled:
                print("\n[INFO] Listening for Android Arena commands via Bluetooth...")
                print("[INFO] Press SP button in Android Arena to start Task 2")
                bluetooth_thread = threading.Thread(target=handler._bluetooth_listen_thread, daemon=True)
                bluetooth_thread.start()
            else:

            # Keep the program running and listening for Android commands

            while True:
